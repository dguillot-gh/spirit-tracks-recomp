# Recompiler hooks for BKIP (Spirit Tracks, Europe).
# Format: <module> <address> replace|before|instead <c_function>
#   replace: the recompiled body is replaced by a call to the C function.
#   before:  the C function is called on entry, then the recompiled body runs.
#   instead: the C function (returning int) is called on entry; the recompiled body runs if it returns 0.
#   at:      the C function is called just before the instruction at <address> (anywhere in a function).
# An optional "=<word>" after the function: the hook applies only where that instruction word is (else left out).
# The decomp leaves these NitroSDK functions unnamed (OS_func_0062..0064, func_01ffbd70); each was matched to Phantom
# Addresses mapped from BKIE's through the zeldaret/st symbol names (config/usa vs config/eur); every function has the USA one's size.
# Hourglass's by size, order and literals (ST-PORT.md).

# NitroSDK thread context switches: guest threads run on host fibers (runtime/src/nitro.cpp).
main 02027120 replace ntr_nitro_save_context   # OS_SaveContext (OS_func_0063)
main 0202716c replace ntr_nitro_load_context   # OS_LoadContext (OS_func_0064)
main 020270b4 replace ntr_nitro_init_context   # OS_InitContext (OS_func_0062)

# Widescreen: the projection's aspect ratio, in NitroSDK's G3i_PerspectiveW_ (r2 = aspect, 0x1555 = 4:3). Unnamed in
# the decomp; matched to Phantom Hourglass's (0x02005bfc): the same arguments, the hardware divider, the same scaling.
main 02022bf4 before st_perspective_aspect      # MTX_PerspectiveW (G3i_PerspectiveW_)
# ...but the camera builds that matrix only when its field of view changes, and loads it every frame: the aspect is
# also set on the way into the geometry engine (st_hooks.cpp).
main 02024a14 instead st_load_mtx44             # G3_LoadMtx44 (the wide projection: the geometry engine's only)
# ...and the camera's view planes, which decide which actors are drawn, take in the wide picture (st_hooks.cpp).
ov000 020764c0 before st_camera_planes           # func_ov000_020764c0: the camera's view planes

# IRQ dispatch: enters OS_IrqHandler_ThreadSwitch explicitly (frontend/src/st_hooks.cpp).
itcm 01ffbd70 replace st_os_irq_handler        # OS_IrqHandler

main 02008b58 before  st_text_obj_init         # NNS_G2dCharCanvasInitForOBJ1D

ov000 02061eb8 before st_text_obj_clear        # UnkSystem2_UnkSubSystem11_Base::vfunc_0C: fills the canvas (next page)
main 020087f8 before  st_text_clear_area       # ClearArea1D
main 02008a0c instead st_text_draw_char        # NNS_G2dCharCanvasDrawChar
ov000 0206216c before st_text_obj_draw         # UnkSystem2_UnkSubSystem11_Derived1: draw (func_ov000_0206216c)
